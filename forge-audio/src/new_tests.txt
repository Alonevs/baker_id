
    #[test]
    fn test_audio_manager_master_volume_and_config() {
        let mut manager = AudioManager::new();
        
        manager.set_master_volume(0.5);
        assert_eq!(manager.mixer.get_master_volume(), 0.5);
        
        manager.set_master_volume(1.5);
        assert_eq!(manager.mixer.get_master_volume(), 1.5);
        
        manager.set_master_volume(-0.5);
        assert_eq!(manager.mixer.get_master_volume(), -0.5);
        
        manager.configure_bus("master_bus", &BusType::Master);
        
        let bus = manager.bus_system.get_bus("master_bus").unwrap();
        assert_eq!(bus.bus_type, BusType::Master);
    }

    #[test]
    fn test_audio_manager_remove_source() {
        let mut manager = AudioManager::new();
        
        let sample1 = AudioSample::new(
            "sfx1".to_string(),
            "SFX 1".to_string(),
            std::path::PathBuf::from("test1.wav")
        );
        let sample2 = AudioSample::new(
            "sfx2".to_string(),
            "SFX 2".to_string(),
            std::path::PathBuf::from("test2.wav")
        );
        
        manager.load_sample(sample1);
        manager.load_sample(sample2);
        
        assert_eq!(manager.get_source_count(), 2);
        
        manager.get_source_mut("sfx1").unwrap().stop();
        manager.sources.remove("sfx1");
        
        assert_eq!(manager.get_source_count(), 1);
        assert!(!manager.sources.contains_key("sfx1"));
        assert!(manager.sources.contains_key("sfx2"));
    }
